' ==========================================================
'  데이터 로드 (Checklist 시트)
' ==========================================================
Private Sub LoadData()
    Dim ws As Worksheet, lastRow As Long, v As Variant
    Dim r As Long, k As Long, i As Long, a As Long, t As Long

    nData = 0: nA = 0: nT = 0
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets("Checklist")
    On Error GoTo 0
    If ws Is Nothing Then Exit Sub

    lastRow = ws.Cells(ws.Rows.Count, "B").End(xlUp).Row
    If lastRow < 2 Then Exit Sub
    v = ws.Range("A2:I" & lastRow).Value2
    k = UBound(v, 1)

    ReDim dRow(1 To k): ReDim dArea(1 To k): ReDim dTask(1 To k)
    ReDim dTime(1 To k): ReDim dTCyc(1 To k): ReDim dSub(1 To k)
    ReDim dCyc(1 To k): ReDim dText(1 To k): ReDim dDone(1 To k)
    ReDim aIdx(1 To k): ReDim tIdx(1 To k)
    ReDim aName(1 To k): ReDim aOpen(1 To k)
    ReDim tName(1 To k): ReDim tArea(1 To k): ReDim tOpen(1 To k)
    ReDim tFirst(1 To k): ReDim tCount(1 To k)

    For r = 1 To k
        If Clean(v(r, 2)) <> "" Then
            nData = nData + 1
            i = nData
            dRow(i) = r + 1
            dArea(i) = Clean(v(r, 2))
            dTask(i) = Clean(v(r, 3))
            dTime(i) = Clean(v(r, 4))
            dTCyc(i) = Clean(v(r, 5))
            dSub(i) = Clean(v(r, 6))
            dCyc(i) = Clean(v(r, 7))
            If dCyc(i) = "" Then dCyc(i) = dTCyc(i)
            If IsError(v(r, 8)) Then dText(i) = "" Else dText(i) = CStr(v(r, 8))
            dDone(i) = v(r, 9)

            ' 업무분야 번호
            a = 0
            If nA > 0 Then
                If aName(nA) = dArea(i) Then a = nA
            End If
            If a = 0 Then nA = nA + 1: a = nA: aName(a) = dArea(i)
            aIdx(i) = a

            ' 업무명 번호
            t = 0
            If nT > 0 Then
                If tArea(nT) = a And tName(nT) = dTask(i) Then t = nT
            End If
            If t = 0 Then
                nT = nT + 1: t = nT
                tName(t) = dTask(i): tArea(t) = a: tFirst(t) = i
            End If
            tIdx(i) = t
            tCount(t) = tCount(t) + 1
        End If
    Next r
    If nA > 0 Then aOpen(1) = True
End Sub

Private Function Clean(ByVal x As Variant) As String
    If IsError(x) Then Exit Function
    If IsEmpty(x) Then Exit Function
    Clean = Trim(Replace(Replace(CStr(x), vbCr, " "), vbLf, " "))
End Function

Private Function IsDone(ByVal i As Long) As Boolean
    If i < 1 Then Exit Function
    If IsDate(dDone(i)) Then IsDone = (Int(CDbl(CDate(dDone(i)))) = CLng(Date))
End Function

Private Function IsSelfTask(ByVal t As Long) As Boolean
    IsSelfTask = (tCount(t) = 1 And dSub(tFirst(t)) = "")
End Function

Private Function AreaDone(ByVal a As Long, ByRef tot As Long) As Long
    Dim i As Long, n As Long
    tot = 0
    For i = 1 To nData
        If aIdx(i) = a Then
            tot = tot + 1
            If IsDone(i) Then n = n + 1
        End If
    Next i
    AreaDone = n
End Function

Private Function TaskDone(ByVal t As Long, ByRef tot As Long) As Long
    Dim i As Long, n As Long
    tot = 0
    For i = 1 To nData
        If tIdx(i) = t Then
            tot = tot + 1
            If IsDone(i) Then n = n + 1
        End If
    Next i
    TaskDone = n
End Function

Private Sub UpdateTotal()
    Dim i As Long, n As Long
    For i = 1 To nData
        If IsDone(i) Then n = n + 1
    Next i
    lblTot.Caption = "오늘 확인완료  " & n & " / " & nData & " 건"
End Sub


' ==========================================================
'  트리(ListBox)
' ==========================================================
Private Function Matches(ByVal i As Long) As Boolean
    Dim key As String, s As String, c As String
    c = cboCyc.Value
    If c <> "" And c <> "점검주기 전체" Then
        If InStr(dCyc(i), c) = 0 Then Exit Function
    End If
    key = LCase(Trim(txtSearch.Text))
    If key <> "" Then
        s = LCase(dArea(i) & "|" & dTask(i) & "|" & dTime(i) & "|" & dSub(i) & "|" & dText(i))
        If InStr(s, key) = 0 Then Exit Function
    End If
    Matches = True
End Function

Private Sub BuildTree()
    Dim a As Long, t As Long, i As Long, nh As Long, ah As Long

    lstTree.Clear
    nodeCount = 0
    If nData = 0 Then Exit Sub
    ReDim nodeKind(1 To nData + nA + nT + 2)
    ReDim nodeRef(1 To nData + nA + nT + 2)
    ReDim nodeLvl(1 To nData + nA + nT + 2)
    searching = (Trim(txtSearch.Text) <> "" Or (cboCyc.ListIndex > 0))

    For a = 1 To nA
        ah = 0
        For i = 1 To nData
            If aIdx(i) = a Then If Matches(i) Then ah = ah + 1
        Next i
        If ah > 0 Then
            AddNode 1, a, 0
            If searching Or aOpen(a) Then
                For t = 1 To nT
                    If tArea(t) = a Then
                        nh = 0
                        For i = 1 To nData
                            If tIdx(i) = t Then If Matches(i) Then nh = nh + 1
                        Next i
                        If nh > 0 Then
                            If IsSelfTask(t) Then
                                AddNode 3, tFirst(t), 1
                            Else
                                AddNode 2, t, 1
                                If searching Or tOpen(t) Then
                                    For i = 1 To nData
                                        If tIdx(i) = t Then If Matches(i) Then AddNode 3, i, 2
                                    Next i
                                End If
                            End If
                        End If
                    End If
                Next t
            End If
        End If
    Next a
End Sub

Private Sub AddNode(ByVal kind As Integer, ByVal ref As Long, ByVal lvl As Integer)
    nodeCount = nodeCount + 1
    nodeKind(nodeCount) = kind
    nodeRef(nodeCount) = ref
    nodeLvl(nodeCount) = lvl
    lstTree.AddItem NodeText(nodeCount)
End Sub

Private Function NodeText(ByVal k As Long) As String
    Dim arrow As String, tot As Long, n As Long, s As String, i As Long
    Select Case nodeKind(k)
    Case 1
        If searching Or aOpen(nodeRef(k)) Then arrow = ChrW(&H25BC) Else arrow = ChrW(&H25B6)
        n = AreaDone(nodeRef(k), tot)
        NodeText = arrow & "  " & aName(nodeRef(k)) & "   (" & n & "/" & tot & ")"
    Case 2
        If searching Or tOpen(nodeRef(k)) Then arrow = ChrW(&H25BC) Else arrow = ChrW(&H25B6)
        n = TaskDone(nodeRef(k), tot)
        NodeText = "      " & arrow & " " & tName(nodeRef(k)) & "   (" & n & "/" & tot & ")"
    Case Else
        i = nodeRef(k)
        If nodeLvl(k) = 1 Then
            s = "      " & ChrW(&H2022) & " " & dTask(i)
        Else
            s = "            " & ChrW(&H2514) & " " & dSub(i)
        End If
        If IsDone(i) Then s = s & "   [완료]"
        NodeText = s
    End Select
End Function

Private Sub RefreshNodeTexts()
    Dim k As Long
    busy = True
    For k = 1 To nodeCount
        lstTree.List(k - 1) = NodeText(k)
    Next k
    busy = False
End Sub

Private Sub lstTree_MouseDown(ByVal Button As Integer, ByVal Shift As Integer, _
                              ByVal X As Single, ByVal Y As Single)
    mouseTick = Timer
End Sub

Private Sub lstTree_Click()
    Dim k As Long
    If busy Then Exit Sub
    k = lstTree.ListIndex + 1
    If k < 1 Or k > nodeCount Then Exit Sub
    If nodeKind(k) < 3 And Abs(Timer - mouseTick) < 1 Then
        mouseTick = -100
        ToggleNode k
    Else
        OnSelect k
    End If
End Sub

Private Sub lstTree_Change()
    If busy Then Exit Sub
    OnSelect lstTree.ListIndex + 1
End Sub

Private Sub lstTree_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    Dim k As Long
    k = lstTree.ListIndex + 1
    If k < 1 Or k > nodeCount Then Exit Sub
    If nodeKind(k) > 2 Then Exit Sub
    Select Case KeyCode
        Case 39: ToggleNode k, 1: KeyCode = 0
        Case 37: ToggleNode k, 0: KeyCode = 0
        Case 13, 32: ToggleNode k: KeyCode = 0
    End Select
End Sub

Private Sub OnSelect(ByVal k As Long)
    If busy Then Exit Sub
    If k < 1 Or k > nodeCount Then
        curRow = 0
        ShowGuide
        Exit Sub
    End If
    Select Case nodeKind(k)
        Case 1: ShowArea nodeRef(k)
        Case 2: ShowTask nodeRef(k)
        Case Else: ShowLeaf nodeRef(k)
    End Select
End Sub

Private Sub ToggleNode(ByVal k As Long, Optional ByVal st As Integer = -1)
    Dim j As Long, topIdx As Long, kind As Integer, ref As Long, nv As Boolean

    kind = nodeKind(k): ref = nodeRef(k)
    If searching Then
        If kind = 1 Then ShowArea ref Else ShowTask ref
        Exit Sub
    End If

    If kind = 1 Then
        If st = -1 Then nv = Not aOpen(ref) Else nv = (st = 1)
        aOpen(ref) = nv
    Else
        If st = -1 Then nv = Not tOpen(ref) Else nv = (st = 1)
        tOpen(ref) = nv
    End If
    topIdx = lstTree.TopIndex

    busy = True
    BuildTree
    For j = 1 To nodeCount
        If nodeKind(j) = kind And nodeRef(j) = ref Then
            lstTree.ListIndex = j - 1
            Exit For
        End If
    Next j
    On Error Resume Next
    lstTree.TopIndex = topIdx
    On Error GoTo 0
    busy = False

    If kind = 1 Then ShowArea ref Else ShowTask ref
End Sub

Private Sub SetAll(ByVal st As Boolean)
    Dim i As Long
    For i = 1 To nA: aOpen(i) = st: Next i
    For i = 1 To nT: tOpen(i) = st: Next i
    busy = True
    BuildTree
    On Error Resume Next
    lstTree.ListIndex = -1
    On Error GoTo 0
    busy = False
    curRow = 0
    ShowGuide
End Sub

Private Sub btnExp_Click()
    SetAll True
End Sub

Private Sub btnCol_Click()
    SetAll False
End Sub

Private Sub btnClr_Click()

    txtSearch.SetFocus
End Sub

Private Sub txtSearch_Change()

End Sub

Private Sub Refilter()
    If busy Then Exit Sub
    busy = True
    BuildTree
    On Error Resume Next
    lstTree.ListIndex = -1
    On Error GoTo 0
    busy = False
    curRow = 0
    ShowGuide
End Sub

Private Sub SetBadges(ByVal cyc As String, ByVal tm As String)
    lblCycle.Caption = IIf(cyc = "", "점검주기 : -", "점검주기 : " & cyc)
    lblCycle.BackStyle = fmBackStyleOpaque
    lblTiming.Caption = tm
    If tm = "" Then
        lblTiming.BackStyle = fmBackStyleTransparent
    Else
        lblTiming.BackStyle = fmBackStyleOpaque
    End If
End Sub

Private Sub ShowArea(ByVal a As Long)
    Dim t As Long, s As String, n As Long, tot As Long
    curRow = 0
    SetLeafMode False
    lblCrumb.Caption = "업무분야"
    lblTitle.Caption = aName(a)
    lblCycle.Caption = "": lblCycle.BackStyle = fmBackStyleTransparent
    lblTiming.Caption = "": lblTiming.BackStyle = fmBackStyleTransparent
    n = AreaDone(a, tot)
    lblHead.Caption = "업무 목록  (오늘 완료 " & n & " / " & tot & ")"
    For t = 1 To nT
        If tArea(t) = a Then
            n = TaskDone(t, tot)
            s = s & tName(t) & "   (" & n & "/" & tot & ")" & vbCrLf
        End If
    Next t
    txtBody.Text = s
End Sub

Private Sub ShowTask(ByVal t As Long)
    Dim i As Long, s As String, n As Long, tot As Long
    curRow = 0
    SetLeafMode False
    lblCrumb.Caption = aName(tArea(t))
    lblTitle.Caption = tName(t)
    i = tFirst(t)
    SetBadges dTCyc(i), dTime(i)
    n = TaskDone(t, tot)
    lblHead.Caption = "세부업무  (오늘 완료 " & n & " / " & tot & ")"
    For i = 1 To nData
        If tIdx(i) = t Then
            s = s & dSub(i)
            If IsDone(i) Then s = s & "   [완료]"
            s = s & vbCrLf
        End If
    Next i
    txtBody.Text = s
End Sub

Private Sub ShowLeaf(ByVal i As Long)
    curRow = i
    SetLeafMode True
    lblCrumb.Caption = aName(aIdx(i)) & "  " & ChrW(&H25B6) & "  " & dTask(i) & IIf(dSub(i) = "", "", "  " & ChrW(&H25B6) & "  " & dSub(i))
    lblTitle.Caption = IIf(dSub(i) = "", dTask(i), dSub(i))
    SetBadges dCyc(i), dTime(i)
    lblHead.Caption = "확인사항"
    txtBody.Text = Replace(dText(i), vbLf, vbCrLf)
    LoadHist i
    lblStatus.Caption = IIf(IsDone(i), "오늘 완료됨 (" & Format(dDone(i), "hh:nn") & ")", "")
End Sub

Private Sub LoadHist(ByVal i As Long)
    Dim ws As Worksheet, r As Long, n As Long, last As Long
    lstHist.Clear
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets("Events")
    On Error GoTo 0
    If ws Is Nothing Then Exit Sub
    last = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    For r = last To 2 Step -1
        If CStr(ws.Cells(r, 3).Value) = dTask(i) And CStr(ws.Cells(r, 4).Value) = dSub(i) And CStr(ws.Cells(r, 2).Value) = dArea(i) Then
            lstHist.AddItem Format(ws.Cells(r, 1).Value, "yyyy-mm-dd hh:nn") & "   " & ws.Cells(r, 5).Value
            n = n + 1
            If n >= 5 Then Exit For
        End If
    Next r
    If n = 0 Then lstHist.AddItem "기록된 점검 이력이 없습니다"
End Sub


' ==========================================================
'  기록 / 취소 / 닫기
' ==========================================================
Private Sub btnRec_Click()
    Dim ws As Worksheet, ev As Worksheet, r As Long, nowT As Date
    If curRow < 1 Then Exit Sub
    nowT = Now
    Set ws = ThisWorkbook.Worksheets("Checklist")
    Set ev = ThisWorkbook.Worksheets("Events")
    r = ev.Cells(ev.Rows.Count, "A").End(xlUp).Row + 1
    ev.Cells(r, 1).Value = nowT
    ev.Cells(r, 1).NumberFormat = "yyyy-mm-dd hh:nn"
    ev.Cells(r, 2).Value = dArea(curRow)
    ev.Cells(r, 3).Value = dTask(curRow)
    ev.Cells(r, 4).Value = dSub(curRow)
    ev.Cells(r, 5).Value = "완료"
    ws.Cells(dRow(curRow), 9).Value = nowT
    ws.Cells(dRow(curRow), 9).NumberFormat = "yyyy-mm-dd hh:nn"
    dDone(curRow) = nowT
    AfterChange
    lblStatus.Caption = "점검 이력을 기록했습니다 (" & Format(nowT, "hh:nn") & ")"
End Sub

Private Sub btnUndo_Click()
    If curRow < 1 Then Exit Sub
    ThisWorkbook.Worksheets("Checklist").Cells(dRow(curRow), 9).ClearContents
    dDone(curRow) = Empty
    AfterChange
    lblStatus.Caption = "완료 표시를 취소했습니다"
End Sub

Private Sub AfterChange()
    RefreshNodeTexts
    UpdateTotal
    LoadHist curRow
End Sub

Private Sub btnClose_Click()
    Unload Me
End Sub
