Private Sub LoadData()
    Dim ws As Worksheet, lastRow As Long, v As Variant
    Dim r As Long, k As Long, i As Long, a As Long, t As Long

    nData = 0: nA = 0: nT = 0
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets("Checklist")
    On Error GoTo 0
    If ws Is Nothing Then Exit Sub

    lastRow = ws.Cells(ws.Rows.Count, "B").End(xlUp).Row
    If lastRow < 2 Then Exit Sub
    v = ws.Range("A2:J" & lastRow).Value2
    k = UBound(v, 1)

    ReDim dRow(1 To k): ReDim dArea(1 To k): ReDim dTask(1 To k)
    ReDim dTime(1 To k): ReDim dTCyc(1 To k): ReDim dSub(1 To k)
    ReDim dCyc(1 To k): ReDim dEff(1 To k): ReDim dText(1 To k)
    ReDim dDone(1 To k): ReDim dMemo(1 To k)
    ReDim aIdx(1 To k): ReDim tIdx(1 To k)
    ReDim aName(1 To k): ReDim aOpen(1 To k)
    ReDim tName(1 To k): ReDim tArea(1 To k): ReDim tOpen(1 To k)
    ReDim tFirst(1 To k): ReDim tCount(1 To k)

    For r = 1 To k
        If Clean(v(r, 2)) <> "" Then
            nData = nData + 1
            i = nData
            dRow(i) = r + 1
            dArea(i) = Clean(v(r, 2))
            dTask(i) = Clean(v(r, 3))
            dTime(i) = Clean(v(r, 4))
            dTCyc(i) = Clean(v(r, 5))
            dSub(i) = Clean(v(r, 6))
            dCyc(i) = Clean(v(r, 7))
            If dCyc(i) = "" Then dCyc(i) = dTCyc(i)
            dEff(i) = EffCycle(dCyc(i))
            If IsError(v(r, 8)) Then dText(i) = "" Else dText(i) = CStr(v(r, 8))
            dDone(i) = v(r, 9)
            If IsError(v(r, 10)) Then dMemo(i) = "" Else dMemo(i) = CStr(v(r, 10))

            a = 0
            If nA > 0 Then
                If aName(nA) = dArea(i) Then a = nA
            End If
            If a = 0 Then nA = nA + 1: a = nA: aName(a) = dArea(i)
            aIdx(i) = a

            t = 0
            If nT > 0 Then
                If tArea(nT) = a And tName(nT) = dTask(i) Then t = nT
            End If
            If t = 0 Then
                nT = nT + 1: t = nT
                tName(t) = dTask(i): tArea(t) = a: tFirst(t) = i
            End If
            tIdx(i) = t
            tCount(t) = tCount(t) + 1
        End If
    Next r
    If nA > 0 Then aOpen(1) = True

    ' 내 업무분야 콤보
    cboArea.AddItem "전체 업무분야"
    For a = 1 To nA
        cboArea.AddItem aName(a)
    Next a
    cboArea.ListIndex = 0
    Dim saved As String
    saved = ""
    On Error Resume Next
    saved = CStr(ws.Range("L1").Value)
    On Error GoTo 0
    If saved <> "" Then
        For a = 1 To nA
            If aName(a) = saved Then cboArea.ListIndex = a
        Next a
    End If
End Sub

Private Sub SaveMyArea()
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("Checklist")
    If cboArea.ListIndex > 0 Then ws.Range("L1").Value = cboArea.Value Else ws.Range("L1").ClearContents
End Sub

Private Function Clean(ByVal x As Variant) As String
    If IsError(x) Then Exit Function
    If IsEmpty(x) Then Exit Function
    Clean = Trim(Replace(Replace(CStr(x), vbCr, " "), vbLf, " "))
End Function

' 여러 주기 중 가장 짧은 주기를 대표 주기로 사용
Private Function EffCycle(ByVal c As String) As String
    If InStr(c, "일") > 0 Then
        EffCycle = "일"
    ElseIf InStr(c, "월") > 0 Then
        EffCycle = "월"
    ElseIf InStr(c, "분기") > 0 Then
        EffCycle = "분기"
    ElseIf InStr(c, "반기") > 0 Then
        EffCycle = "반기"
    ElseIf InStr(c, "연간") > 0 Then
        EffCycle = "연간"
    Else
        EffCycle = "수시"
    End If
End Function

Private Function EffIdx(ByVal e As String) As Long
    Select Case e
        Case "일": EffIdx = 1
        Case "월": EffIdx = 2
        Case "분기": EffIdx = 3
        Case "반기": EffIdx = 4
        Case "연간": EffIdx = 5
        Case Else: EffIdx = 6
    End Select
End Function

Private Function GroupLabel(ByVal g As Long) As String
    Select Case g
        Case 1: GroupLabel = "매일"
        Case 2: GroupLabel = "매월"
        Case 3: GroupLabel = "분기별"
        Case 4: GroupLabel = "반기별"
        Case 5: GroupLabel = "연간"
        Case Else: GroupLabel = "수시 / 기타"
    End Select
End Function

Private Function PeriodName(ByVal e As String) As String
    Select Case e
        Case "일": PeriodName = "오늘"
        Case "월": PeriodName = "이번 달"
        Case "분기": PeriodName = "이번 분기"
        Case "반기": PeriodName = "이번 반기"
        Case "연간": PeriodName = "올해"
        Case Else: PeriodName = "오늘"
    End Select
End Function

' 기간 키 : 같은 키이면 '같은 기간'
Private Function PeriodKey(ByVal e As String, ByVal d As Date) As Long
    Select Case e
        Case "월": PeriodKey = Year(d) * 100 + Month(d)
        Case "분기": PeriodKey = Year(d) * 10 + ((Month(d) - 1) \ 3 + 1)
        Case "반기": PeriodKey = Year(d) * 10 + IIf(Month(d) <= 6, 1, 2)
        Case "연간": PeriodKey = Year(d)
        Case Else: PeriodKey = CLng(Int(CDbl(d)))
    End Select
End Function

' 이번 기간에 이미 완료했는가?
Private Function IsDone(ByVal i As Long) As Boolean
    If i < 1 Then Exit Function
    If IsEmpty(dDone(i)) Then Exit Function
    If Not IsNumeric(dDone(i)) Then Exit Function
    IsDone = (PeriodKey(dEff(i), CDate(CDbl(dDone(i)))) = PeriodKey(dEff(i), Date))
End Function

Private Function AreaDone(ByVal a As Long, ByRef tot As Long) As Long
    Dim i As Long, n As Long
    tot = 0
    For i = 1 To nData
        If aIdx(i) = a Then
            tot = tot + 1
            If IsDone(i) Then n = n + 1
        End If
    Next i
    AreaDone = n
End Function

Private Function TaskDone(ByVal t As Long, ByRef tot As Long) As Long
    Dim i As Long, n As Long
    tot = 0
    For i = 1 To nData
        If tIdx(i) = t Then
            tot = tot + 1
            If IsDone(i) Then n = n + 1
        End If
    Next i
    TaskDone = n
End Function

Private Function CycDone(ByVal g As Long, ByRef tot As Long) As Long
    Dim i As Long, n As Long
    tot = 0
    For i = 1 To nData
        If EffIdx(dEff(i)) = g And InScope(i) Then
            tot = tot + 1
            If IsDone(i) Then n = n + 1
        End If
    Next i
    CycDone = n
End Function

Private Function InScope(ByVal i As Long) As Boolean
    InScope = True
    If cboArea.ListIndex > 0 Then InScope = (dArea(i) = cboArea.Value)
End Function

Private Sub UpdateTotal()
    Dim i As Long, n As Long, tot As Long, pct As Double
    For i = 1 To nData
        If dEff(i) <> "수시" And InScope(i) Then
            tot = tot + 1
            If IsDone(i) Then n = n + 1
        End If
    Next i
    If tot > 0 Then pct = n / tot
    lblTot.Caption = "이번 기간 완료  " & n & " / " & tot & " 건  (" & Format(pct, "0%") & ")"
    lblBarFill.Width = 340 * pct
    If lblBarFill.Width < 1 Then lblBarFill.Width = 1
End Sub


' ==========================================================
'  트리(ListBox)
' ==========================================================
Private Function Matches(ByVal i As Long) As Boolean
    Dim key As String, s As String
    If Not InScope(i) Then Exit Function
    key = LCase(Trim(txtSearch.Text))
    If key <> "" Then
        s = LCase(dArea(i) & "|" & dTask(i) & "|" & dTime(i) & "|" & dSub(i) & "|" & dText(i))
        If InStr(s, key) = 0 Then Exit Function
    End If
    Matches = True
End Function

Private Sub BuildTree()
    Dim a As Long, t As Long, g As Long, i As Long, nh As Long, ah As Long, pass As Long

    lstTree.Clear
    nodeCount = 0
    If nData = 0 Then Exit Sub
    ReDim nodeKind(1 To nData + nA + nT + 10)
    ReDim nodeRef(1 To nData + nA + nT + 10)
    ReDim nodeLvl(1 To nData + nA + nT + 10)
    searching = (Trim(txtSearch.Text) <> "")

    If viewMode = 0 Then
        For g = 1 To 6
            nh = 0
            For i = 1 To nData
                If EffIdx(dEff(i)) = g Then If Matches(i) Then nh = nh + 1
            Next i
            If nh > 0 Then
                AddNode 1, g, 0
                If searching Or gOpen(g) Then
                    For pass = 0 To 1                       ' 미완료 먼저, 완료는 아래로
                        For i = 1 To nData
                            If EffIdx(dEff(i)) = g Then
                                If Matches(i) Then
                                    If (IsDone(i) And pass = 1) Or (Not IsDone(i) And pass = 0) Then AddNode 3, i, 1
                                End If
                            End If
                        Next i
                    Next pass
                End If
            End If
        Next g
    Else
        For a = 1 To nA
            ah = 0
            For i = 1 To nData
                If aIdx(i) = a Then If Matches(i) Then ah = ah + 1
            Next i
            If ah > 0 Then
                AddNode 1, a, 0
                If searching Or aOpen(a) Then
                    For t = 1 To nT
                        If tArea(t) = a Then
                            nh = 0
                            For i = 1 To nData
                                If tIdx(i) = t Then If Matches(i) Then nh = nh + 1
                            Next i
                            If nh > 0 Then
                                If tCount(t) = 1 And dSub(tFirst(t)) = "" Then
                                    AddNode 3, tFirst(t), 1
                                Else
                                    AddNode 2, t, 1
                                    If searching Or tOpen(t) Then
                                        For i = 1 To nData
                                            If tIdx(i) = t Then If Matches(i) Then AddNode 3, i, 2
                                        Next i
                                    End If
                                End If
                            End If
                        End If
                    Next t
                End If
            End If
        Next a
    End If
End Sub

Private Sub AddNode(ByVal kind As Integer, ByVal ref As Long, ByVal lvl As Integer)
    nodeCount = nodeCount + 1
    nodeKind(nodeCount) = kind
    nodeRef(nodeCount) = ref
    nodeLvl(nodeCount) = lvl
    lstTree.AddItem NodeText(nodeCount)
End Sub

Private Function NodeText(ByVal k As Long) As String
    Dim arrow As String, tot As Long, n As Long, s As String, i As Long, isOpen As Boolean
    Select Case nodeKind(k)
    Case 1
        If viewMode = 0 Then
            isOpen = searching Or gOpen(nodeRef(k))
            n = CycDone(nodeRef(k), tot)
            s = GroupLabel(nodeRef(k))
        Else
            isOpen = searching Or aOpen(nodeRef(k))
            n = AreaDone(nodeRef(k), tot)
            s = aName(nodeRef(k))
        End If
        If isOpen Then arrow = ChrW(&H25BC) Else arrow = ChrW(&H25B6)
        NodeText = arrow & "  " & s & "   (" & n & "/" & tot & ")"
    Case 2
        If searching Or tOpen(nodeRef(k)) Then arrow = ChrW(&H25BC) Else arrow = ChrW(&H25B6)
        n = TaskDone(nodeRef(k), tot)
        NodeText = "      " & arrow & " " & tName(nodeRef(k)) & "   (" & n & "/" & tot & ")"
    Case Else
        i = nodeRef(k)
        If viewMode = 0 Then
            s = "      [" & Left(dArea(i), 2) & "] " & dTask(i)
            If dSub(i) <> "" Then s = s & " " & ChrW(&H203A) & " " & dSub(i)
        ElseIf nodeLvl(k) = 1 Then
            s = "      " & ChrW(&H2022) & " " & dTask(i)
        Else
            s = "            " & ChrW(&H2514) & " " & dSub(i)
        End If
        If IsDone(i) Then s = s & "   [완료]"
        NodeText = s
    End Select
End Function

Private Sub RefreshNodeTexts()
    Dim k As Long
    busy = True
    For k = 1 To nodeCount
        lstTree.List(k - 1) = NodeText(k)
    Next k
    busy = False
End Sub

Private Sub lstTree_Click()
    Dim k As Long
    If busy Then Exit Sub
    k = lstTree.ListIndex + 1
    If k < 1 Or k > nodeCount Then Exit Sub
    If nodeKind(k) < 3 And Abs(Timer - mouseTick) < 1 Then

        ToggleNode k
    Else

    End If
End Sub

Private Sub lstTree_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    Dim k As Long
    k = lstTree.ListIndex + 1
    If k < 1 Or k > nodeCount Then Exit Sub
    If nodeKind(k) > 2 Then Exit Sub
    Select Case KeyCode
        Case 39: ToggleNode k, 1: KeyCode = 0
        Case 37: ToggleNode k, 0: KeyCode = 0
        Case 13, 32: ToggleNode k: KeyCode = 0
    End Select
End Sub

Private Sub OnSelect(ByVal k As Long)
    If busy Then Exit Sub
    If k < 1 Or k > nodeCount Then
        ShowGuide
        Exit Sub
    End If
    Select Case nodeKind(k)
        Case 1
            If viewMode = 0 Then ShowCycGroup nodeRef(k) Else ShowArea nodeRef(k)
        Case 2: ShowTask nodeRef(k)
        Case Else: ShowLeaf nodeRef(k)
    End Select
End Sub

Private Sub ToggleNode(ByVal k As Long, Optional ByVal st As Integer = -1)
    Dim j As Long, topIdx As Long, kind As Integer, ref As Long, nv As Boolean

    kind = nodeKind(k): ref = nodeRef(k)
    If searching Then
        OnSelect k
        Exit Sub
    End If

    If kind = 1 Then
        If viewMode = 0 Then
            If st = -1 Then nv = Not gOpen(ref) Else nv = (st = 1)
            gOpen(ref) = nv
        Else
            If st = -1 Then nv = Not aOpen(ref) Else nv = (st = 1)
            aOpen(ref) = nv
        End If
    Else
        If st = -1 Then nv = Not tOpen(ref) Else nv = (st = 1)
        tOpen(ref) = nv
    End If
    topIdx = lstTree.TopIndex

    busy = True
    BuildTree
    For j = 1 To nodeCount
        If nodeKind(j) = kind And nodeRef(j) = ref Then
            lstTree.ListIndex = j - 1
            Exit For
        End If
    Next j
    On Error Resume Next
    lstTree.TopIndex = topIdx
    On Error GoTo 0
    busy = False

    OnSelect lstTree.ListIndex + 1
End Sub

    busy = True
    BuildTree
    On Error Resume Next
    lstTree.ListIndex = -1

    ShowGuide
End Sub

        End If

        lblState.Caption = ChrW(&H2714) & " " & PeriodName(dEff(i)) & " 완료됨   " & lastTxt

' 확인사항에서 시스템 메뉴 경로(예: KESCOIN > 사업관리 > ...)를 찾는다
Private Sub FindPaths(ByVal i As Long)
    Dim ln As Variant, s As String
    pathText = ""
    For Each ln In Split(dText(i), vbLf)
        s = Trim(CStr(ln))
        If Left(s, 2) = ChrW(&H25CB) & " " And InStr(s, " > ") > 0 Then
            If pathText <> "" Then pathText = pathText & vbCrLf
            pathText = pathText & Mid(s, 3)
        End If
    Next ln
    btnPath.Visible = (pathText <> "")
End Sub

Private Sub LoadHist(ByVal i As Long)
    Dim ws As Worksheet, r As Long, n As Long, last As Long, memo As String
    lstHist.Clear
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets("Events")
    On Error GoTo 0
    If ws Is Nothing Then Exit Sub
    last = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    For r = last To 2 Step -1
        If CStr(ws.Cells(r, 3).Value) = dTask(i) And CStr(ws.Cells(r, 4).Value) = dSub(i) And CStr(ws.Cells(r, 2).Value) = dArea(i) Then
            memo = CStr(ws.Cells(r, 6).Value)
            lstHist.AddItem Format(ws.Cells(r, 1).Value, "yyyy-mm-dd hh:nn") & "   " & ws.Cells(r, 5).Value & IIf(memo = "", "", "  - " & memo)
            n = n + 1
            If n >= 5 Then Exit For
        End If
    Next r
    If n = 0 Then lstHist.AddItem "기록된 점검 이력이 없습니다"
End Sub

Private Sub AfterChange()
    Dim k As Long, keep As Long
    keep = lstTree.ListIndex
    busy = True
    BuildTree                     ' 완료 항목은 아래쪽으로 재정렬
    If keep >= 0 And keep < nodeCount Then
        For k = 1 To nodeCount
            If nodeKind(k) = 3 And nodeRef(k) = curRow Then lstTree.ListIndex = k - 1: Exit For
        Next k
    End If
    busy = False
    UpdateTotal
    ShowState curRow
    LoadHist curRow
End Sub
